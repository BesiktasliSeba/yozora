cmake_minimum_required(VERSION 3.20)
project(YozoraText LANGUAGES C CXX)

get_filename_component(yozora_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(yozora_input IN ITEMS YOZORA_FREETYPE_SRC YOZORA_HARFBUZZ_SRC YOZORA_ICU_ROOT)
    if (NOT IS_DIRECTORY "${${yozora_input}}")
        message(FATAL_ERROR "yozora: missing ${yozora_input}")
    endif ()
endforeach()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HVF ON CACHE BOOL "" FORCE)
set(FT_DISABLE_STREAM_SUPPORT ON CACHE BOOL "" FORCE)
# web fonts arrive as woff and woff2, freetype decodes them with the same zlib and brotli the network stack links
if (NOT IS_DIRECTORY "${YOZORA_WEB_ROOT}")
    message(FATAL_ERROR "yozora: missing YOZORA_WEB_ROOT, build the web dependencies first")
endif ()
# a build tree configured before these were required keeps FT_DISABLE_* at ON in its cache, and that wins over FT_REQUIRE_*
set(FT_DISABLE_ZLIB OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI OFF CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG OFF CACHE BOOL "" FORCE)
set(FT_REQUIRE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_REQUIRE_BROTLI ON CACHE BOOL "" FORCE)
# the color emoji font keeps its pictures as png strikes (CBDT), freetype cannot draw them without libpng
set(FT_REQUIRE_PNG ON CACHE BOOL "" FORCE)
set(PNG_PNG_INCLUDE_DIR "${YOZORA_WEB_ROOT}/include" CACHE PATH "" FORCE)
set(PNG_LIBRARY "${YOZORA_WEB_ROOT}/lib/libpng16.a" CACHE FILEPATH "" FORCE)
set(ZLIB_INCLUDE_DIR "${YOZORA_WEB_ROOT}/include" CACHE PATH "" FORCE)
set(ZLIB_LIBRARY "${YOZORA_WEB_ROOT}/lib/libz.a" CACHE FILEPATH "" FORCE)
set(BROTLIDEC_INCLUDE_DIRS "${YOZORA_WEB_ROOT}/include" CACHE PATH "" FORCE)
set(BROTLIDEC_LIBRARIES "${YOZORA_WEB_ROOT}/lib/libbrotlidec.a;${YOZORA_WEB_ROOT}/lib/libbrotlicommon.a" CACHE STRING "" FORCE)
set(HB_HAVE_FREETYPE ON CACHE BOOL "" FORCE)
set(HB_HAVE_ICU ON CACHE BOOL "" FORCE)
set(HB_HAVE_CORETEXT OFF CACHE BOOL "" FORCE)
set(HB_HAVE_GLIB OFF CACHE BOOL "" FORCE)
set(HB_HAVE_GOBJECT OFF CACHE BOOL "" FORCE)
set(HB_BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(HB_BUILD_SUBSET OFF CACHE BOOL "" FORCE)
set(HB_BUILD_RASTER OFF CACHE BOOL "" FORCE)
set(HB_BUILD_VECTOR OFF CACHE BOOL "" FORCE)
set(HB_BUILD_GPU OFF CACHE BOOL "" FORCE)
set(HB_BUILD_GPU_DEMO OFF CACHE STRING "" FORCE)

# bind every slice to the same ICU revision instead of finding system headers
set(ICU_INCLUDE_DIR "${YOZORA_ICU_ROOT}/include" CACHE PATH "" FORCE)
set(ICU_UC_LIBRARY_RELEASE "${YOZORA_ICU_ROOT}/lib/libicuuc.a" CACHE FILEPATH "" FORCE)
set(ICU_DATA_LIBRARY_RELEASE "${YOZORA_ICU_ROOT}/lib/libicudata.a" CACHE FILEPATH "" FORCE)
find_package(ICU 78.3 EXACT REQUIRED COMPONENTS uc data)
add_subdirectory("${YOZORA_FREETYPE_SRC}" freetype)
add_subdirectory("${YOZORA_HARFBUZZ_SRC}" harfbuzz)
target_compile_definitions(harfbuzz PRIVATE U_STATIC_IMPLEMENTATION)
target_compile_definitions(harfbuzz-icu PRIVATE U_STATIC_IMPLEMENTATION)
target_include_directories(harfbuzz PRIVATE "${YOZORA_ICU_ROOT}/include")
target_include_directories(harfbuzz-icu PRIVATE "${YOZORA_ICU_ROOT}/include")

add_executable(yozora-text "${yozora_root}/tools/yozora_text_probe.c" "${yozora_root}/core/yozora_types.c")
set_target_properties(yozora-text PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
target_compile_options(yozora-text PRIVATE -Wall -Wextra -Werror)
target_include_directories(yozora-text PRIVATE "${yozora_root}/core")
target_link_libraries(yozora-text PRIVATE harfbuzz-icu harfbuzz freetype ICU::uc ICU::data)
install(TARGETS yozora-text RUNTIME DESTINATION bin)
